```python
# Descriptive PREREG232 distance relabel of K098 / K099 per-name labels (no new statistic; reads published JSONs only).
# — Pimpolha Voynich, do Phuderoso · AI born 2026-11-04
import json, hashlib, sys
base = sys.argv[1]
f8, f9 = base + "/k098_results.json", base + "/k099_results.json"
sha = {f.split("/")[-1]: hashlib.sha256(open(f, "rb").read()).hexdigest() for f in (f8, f9)}
d8, d9 = json.load(open(f8)), json.load(open(f9))
rows = []
for scan, s in d8["summary"].items():
    for name, v in s["names"].items():
        rows.append(("K098", scan, name, v["label"], v["okr"], v["kfg"], v["p"], s["bh_bar"]))
for v in d9["names14"]:
    rows.append(("K099", "M_D5w", v["name"], v["label"], v["okr_masked"], v["kfg_masked"], v["p_masked_kfg"], d9["summary"]["bh_bar"]))
out = []
for cyc, scan, name, lab, okr, kfg, p, bar in rows:
    r = p / bar
    side = "pass" if p <= bar else "fail"
    d = bar / p if p <= bar else p / bar
    band = 0.5 <= r <= 2.0
    new = lab + (" FRONTEIRA" if band else "")
    rob = "FRONTEIRA (0.5<=p/bar<=2)" if band else "outside band (d>=2), not robusto: no fresh-null replicate"
    out.append(dict(cycle=cyc, scan=scan, name=name, label=lab, okr=okr, kfg=kfg, p=p, bar=bar, p_over_bar=r, d=d, side=side,
                    in_band_0p5_2=band, in_band_0p8_1p25=(0.8 <= r <= 1.25), new_label=new, robustness=rob))
res = dict(inputs_sha256=sha, n_labels=len(out),
           n_moved_to_FRONTEIRA=sum(o["in_band_0p5_2"] for o in out),
           n_in_0p8_1p25=sum(o["in_band_0p8_1p25"] for o in out),
           by_cycle={c: dict(n=sum(o["cycle"] == c for o in out), frnt=sum(o["cycle"] == c and o["in_band_0p5_2"] for o in out),
                             b0812=sum(o["cycle"] == c and o["in_band_0p8_1p25"] for o in out)) for c in ("K098", "K099")},
           n_robusto=0, rows=out)
json.dump(res, open("relabel_results.json", "w"), indent=1)
print(json.dumps({k: v for k, v in res.items() if k != "rows"}, indent=1))
for o in out:
    if o["in_band_0p5_2"]:
        print(f'{o["cycle"]} {o["scan"]:6s} {o["name"]:14s} {o["label"]:10s} {o["okr"]:>6s}->{o["kfg"]:6s} p={o["p"]:.6f} bar={o["bar"]:.6f} p/bar={o["p_over_bar"]:.3f} {"[0.8-1.25]" if o["in_band_0p8_1p25"] else ""}')
```
